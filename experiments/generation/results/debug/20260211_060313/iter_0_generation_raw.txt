import torch
import triton
import triton.language as tl

@triton.jit
def rmsnorm_forward_kernel(
    x_ptr, output_ptr, weight_ptr,
    stride, n_cols, eps,
    BLOCK_SIZE: tl.constexpr,
):
    # Map program id to row index
    row_idx = tl.program_id(0)
    # Compute pointers for this row
    row_ptr = x_ptr + row_idx * stride
    output_row_ptr = output_ptr + row_idx * stride
    
    # Compute mean of squares
    x2sum = 0.0
    col_offsets = tl.arange(0, BLOCK_SIZE)
    for start_n in range(0, n_cols, BLOCK_SIZE):
        # Load block of x
        mask = col_offsets < n_cols - start_n
        x = tl.load(row_ptr + start_n + col_offsets, mask=mask, other=0.0)
        x2sum += tl.sum(tl.pow(x, 2))

    mean_x2 = x2sum / n_cols
    rstd = tl.rsqrt(mean_x2 + eps)

    # Scale and write output
    for start_n in range(0, n_cols, BLOCK_SIZE):
        mask = col_offsets < n_cols - start_n
        x = tl.load(row_ptr + start_n + col_offsets, mask=mask, other=0.0)
        w = tl.load(weight_ptr + start_n + col_offsets, mask=mask, other=0.0)
        
        output = x * rstd * w
        tl.store(output_row_ptr + start_n + col_offsets, output, mask=mask)

@triton.jit
def rmsnorm_backward_dx_kernel(
    x_ptr, grad_output_ptr, grad_x_ptr, weight_ptr,
    stride, n_cols, eps,
    BLOCK_SIZE: tl.constexpr,
):
    row_idx = tl.program_id(0)
    row_ptr = x_ptr + row_idx * stride
    grad_row_ptr = grad_output_ptr + row_idx * stride
    grad_x_row_ptr = grad_x_ptr + row_idx * stride
    
    col_offsets = tl.arange(0, BLOCK_SIZE)

    # 1. Compute rstd from original x
    x2sum = 0.0
    for start_n in range(0, n_cols, BLOCK_SIZE):
        mask = col_offsets < n_cols - start_n
        x = tl.load(row_ptr + start_n + col_offsets, mask=mask, other=0.0)
        x2sum += tl.sum(tl.pow(x, 2))
    
    mean_x2 = x2sum / n_cols
    rstd = tl.rsqrt(mean_x2 + eps)

    # 2. Compute dot(x, dy * w)
    dot_xv = 0.0
    for start_n in range(0, n_cols, BLOCK_SIZE):
        mask = col_offsets < n_cols - start_n
        x = tl.load(row_ptr + start_n + col_offsets, mask=mask, other=0.0)
        dy = tl.load(grad_row_ptr + start_n + col_offsets, mask=mask, other=0.0)
        w = tl.load(weight_ptr + start_n + col_offsets, mask=mask, other=0.0)
        
        v = dy * w
        dot_xv += tl.sum(x * v)

    # 3. Compute gradient
    factor = tl.pow(rstd, 3) * dot_xv / n_cols
    for start_n in range(0, n_cols, BLOCK_SIZE):
        mask = col_offsets < n_cols - start_n
        x = tl.load(row_ptr + start_n + col_offsets, mask=mask, other=0.0)
        dy = tl.load(grad_row_ptr + start_n + col_offsets, mask=mask, other=0.0)
        w = tl.load(weight_ptr + start_n + col_offsets, mask=mask, other=0.0)
        
        v = dy * w
        dx = v * rstd - x * factor
        
        tl.store(grad_x_row_ptr + start_n + col_offsets, dx, mask=mask)

class RMSNormFunction(torch.autograd.Function):
    @staticmethod
    def forward(ctx, x, weight, eps):
        original_shape = x.shape
        x_reshaped = x.reshape(-1, x.shape[-1]).contiguous()
        M, N = x_reshaped.shape
        
        output = torch.empty_like(x_reshaped)
        
        BLOCK_SIZE = 1024
        grid = (M,)
        
        rmsnorm_forward_kernel[grid](
            x_reshaped, output, weight,
            x_reshaped.stride(0), N, eps,
            BLOCK_SIZE=BLOCK_SIZE,
        )
        
        ctx.save_for_backward(x_reshaped, weight)
        ctx.eps = eps
        ctx.original_shape = original_shape
        
        return output.reshape(original_shape)

    @staticmethod
    def backward(ctx, grad_output):
        x_reshaped, weight = ctx.saved_tensors
        eps = ctx.eps
        original_shape = ctx.original_shape
        
        grad_output_reshaped = grad_output.reshape(-1, grad_output.shape[-1]).contiguous()
        M, N = x_reshaped.shape
        
        grad_x = torch.empty_like(x_reshaped)
        grad_w = torch.zeros_like(weight)
        
        BLOCK_SIZE = 1024
        grid = (M,)
