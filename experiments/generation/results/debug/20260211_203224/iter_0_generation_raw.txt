def _reference_safe_softmax(x):

    max_vals = x.max(dim=-1, keepdim=True)[0]

        )
        
        ctx.save_for_backward(output)
        ctx.original_shape = original_shape
        ctx.K = K
        
        return output.reshape(original_shape)

    @staticmethod
    def backward(ctx, grad_output):
        y_reshaped = ctx.saved_tensors[0]
        original_shape = ctx.original_shape
        K = ctx.K
        
        grad_reshaped = grad_output.reshape(-1, grad_output.shape[-1]).contiguous()
        
        dx = torch.empty_like(y_reshaped)
        
        BLOCK_SIZE = 2048
        grid = lambda meta: (y_reshaped.shape[0],)
        
        softmax_backward_kernel[grid](
            y_reshaped, grad_reshaped, dx, y_reshaped.stride(0), K, BLOCK_SIZE=BLOCK_SIZE
        )
        
        return dx.reshape(original_shape)

def triton_softmax(x):
    return MyFunction.apply(x)
