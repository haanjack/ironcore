    sum_y_dy = 0.0
    for off in range(0, n_cols, BLOCK_SIZE):
        col_offsets = off + offsets
        mask = col_offsets < n_cols
        y = tl.load(row_start_y + col_offsets, mask=mask, other=0.0)
        dy = tl.load(row_start_dy + col_offsets, mask=mask, other=0.0)
        y = y.to(tl.float32)
        dy = dy.to(tl.float32)
        sum_y_dy += tl.sum(y * dy)
        
    # Pass 2: Compute dx and write
    row_start_dx = grad_input_ptr + pid * stride
    for off in range(0, n_cols, BLOCK_SIZE):
        col_offsets = off + offsets
        mask = col_offsets < n_cols
        y = tl.load(row_start_y + col_offsets, mask=mask, other=0.0)
        dy = tl.load(row_start_dy + col_offsets, mask=mask, other=0.0)
        y = y.to(tl.float32)
        dy = dy.to(tl.float32)
        dx = y * (dy - sum_y_dy)
        tl.store(row_start_dx + col_offsets, dx)

class _TritonSoftmax(torch.autograd.Function):
    @staticmethod
    def forward(ctx, x):
        # Reshape to 2D (M, K)
        original_shape = x.shape
        x_reshaped = x.reshape(-1, x.shape[-1]).contiguous()
        M, K = x_reshaped.shape
        
        output = torch.empty_like(x_reshaped)
        
        BLOCK_SIZE = 128
        grid = (M,)
        
        # Launch forward kernel
        softmax_forward_kernel[grid](
            x_reshaped, output,
            x_reshaped.stride(0), K,
            BLOCK_SIZE=BLOCK_SIZE
        )
        
        ctx.save_for_backward(output)
        ctx.K = K
        ctx.BLOCK_SIZE = BLOCK_SIZE
        ctx.stride = x_reshaped.stride(0)
        return output.reshape_as(x)

    @staticmethod
    def backward(ctx, grad_output):
        output, = ctx.saved_tensors
        K = ctx.K
        BLOCK_SIZE = ctx.BLOCK_SIZE
        stride = ctx.stride
        
        grad_output_reshaped = grad_output.reshape(-1, K).contiguous()
        grad_input = torch.empty_like(output)
        
        grid = (output.shape[0],)
        
        # Launch backward kernel
        softmax_backward_kernel[grid](
            output, grad_output_reshaped, grad_input,
            stride, K,
            BLOCK_SIZE=BLOCK_SIZE
        )
        
        return grad_input.reshape_as(grad_output)
