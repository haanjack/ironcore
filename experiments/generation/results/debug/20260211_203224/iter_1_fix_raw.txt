@triton.jit
def softmax_forward_kernel(x_ptr, output_ptr, stride, n_cols, BLOCK_SIZE: tl.constexpr):
    pid = tl.program_id(0)
    offsets = tl.arange(0, BLOCK_SIZE)
    
    row_start_ptr = x_ptr + pid * stride
    
    # Pass 1: Find max
    # Use -inf for masked values so they don't affect max
    m = -float('inf')
    for off in range(0, n_cols, BLOCK_SIZE):
        col_offsets = off + offsets
        mask = col_offsets < n_cols
        x = tl.load(row_start_ptr + col_offsets, mask=mask, other=-float('inf'))
        m = tl.maximum(m, tl.max(x))
        
    # Pass 2: Find sum of exps
    # Use -inf for masked values so exp(-inf) = 0, preventing garbage additions to the sum
    l = 0.0
    for off in range(0, n_cols, BLOCK_SIZE):

        x = tl.load(row_start_ptr + col_offsets, mask=mask, other=-float('inf'))
