import torch
import triton
import triton.language as tl

@triton.jit
def forward_kernel(x_ptr, w_ptr, y_ptr, inv_std_ptr, stride, n_rows, n_cols, eps, BLOCK_SIZE: tl.constexpr):
    row = tl.program_id(0)
    col_offsets = tl.arange(0, BLOCK_SIZE)
    
    # Pointer offsets
    x_ptr += row * stride
    y_ptr += row * stride
    
    # Compute mean of squares
    ss = 0.0
    for off in range(0, n_cols, BLOCK_SIZE):
        cols = off + col_offsets
        mask = cols < n_cols
        x = tl.load(x_ptr + cols, mask=mask, other=0.).to(tl.float32)
        ss += tl.sum(tl.pow(x, 2))
    
    mean = ss / n_cols
    inv_std = tl.rsqrt(mean + eps)
    
    # Save inverse std for backward
    tl.store(inv_std_ptr + row, inv_std)
    
    # Normalize and scale
    for off in range(0, n_cols, BLOCK_SIZE):
        cols = off + col_offsets
        mask = cols < n_cols
        x = tl.load(x_ptr + cols, mask=mask, other=0.)
        w = tl.load(w_ptr + cols, mask=mask, other=0.)
        
        y = (x.to(tl.float32) * inv_std) * w
        tl.store(y_ptr + cols, y, mask=mask)

@triton.jit
def backward_kernel(dx_ptr, dw_ptr, dy_ptr, x_ptr, w_ptr, inv_std_ptr, stride, n_rows, n_cols, BLOCK_SIZE: tl.constexpr):
    row = tl.program_id(0)
    col_offsets = tl.arange(0, BLOCK_SIZE)
    
    # Pointer offsets
    x_ptr += row * stride
    dy_ptr += row * stride
    dx_ptr += row * stride
    
    inv_std = tl.load(inv_std_ptr + row).to(tl.float32)
    
    # Compute dsum = sum(x * dy * w)
    dsum = 0.0
    for off in range(0, n_cols, BLOCK_SIZE):
        cols = off + col_offsets
        mask = cols < n_cols
        x = tl.load(x_ptr + cols, mask=mask, other=0.).to(tl.float32)
        dy = tl.load(dy_ptr + cols, mask=mask, other=0.).to(tl.float32)
        w = tl.load(w_ptr + cols, mask=mask, other=0.).to(tl.float32)
        
        dsum += tl.sum(x * dy * w)
        
    # Coefficient for the second term in dx calculation
    # dx = (dy * w) / std - x * (dsum / (N * std^3))
    # coeff = dsum / (N * inv_std^3)
    coeff = dsum / (n_cols * tl.pow(inv_std, 3))
    
    for off in range(0, n_cols, BLOCK_SIZE):
        cols = off + col_offsets
        mask = cols < n_cols
        x = tl.load(x_ptr + cols, mask=mask, other=0.)
        dy = tl.load(dy_ptr + cols, mask=mask, other=0.)

        term1 = dy.to(tl.float32) * w.to(tl.float32) * inv_std
        term2 = x.to(tl.float32) * coeff
        dx = term1 - term2

        dw = dy.to(tl.float32) * x.to(tl.float32) * inv_std
        tl.atomic_add(dw_ptr + cols, dw, mask=mask)
