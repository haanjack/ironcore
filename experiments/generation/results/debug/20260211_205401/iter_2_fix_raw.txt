        ctx.save_for_backward(output)
        ctx.BLOCK_SIZE = BLOCK_SIZE
        ctx.n_cols = n_cols
        ctx.stride = x_reshaped.stride(0)
        
        return output.reshape_as(x)

    @staticmethod
    def backward(ctx, grad_output):
        output, = ctx.saved_tensors
        
        grad_output_reshaped = grad_output.reshape(-1, grad_output.shape[-1]).contiguous()
        n_rows = grad_output_reshaped.shape[0]
        
        grad_input = torch.empty_like(output)
        
        grid = (n_rows,)
        num_warps = 4 if ctx.BLOCK_SIZE < 128 else 8
        
        backward_kernel[grid](
            grad_output_reshaped,
            output,
            grad_input,
            ctx.stride,
            ctx.n_cols,
            BLOCK_SIZE=ctx.BLOCK_SIZE,
            num_warps=num_warps
        )
        
        return grad_input.reshape_as(grad_output)
