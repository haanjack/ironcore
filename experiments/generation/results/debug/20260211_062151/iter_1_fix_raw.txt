def cross_entropy_fwd_kernel(

    stride_logits,
    n_cols,
    ignore_index,
    BLOCK_SIZE: tl.constexpr

    row_logits_ptr = logits_ptr + pid * stride_logits

    for i in range(0, n_cols, BLOCK_SIZE):
