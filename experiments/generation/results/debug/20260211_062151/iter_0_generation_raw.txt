def _reference_softmax_cross_entropy(logits, labels, ignore_index=-1):
    """PyTorch reference: Cross entropy with softmax."""
    # Stable softmax
    logits_max = logits.max(dim=-1, keepdim=True)[0]
    logits_stable = logits - logits_max

    exp_logits = torch.exp(logits_stable)
    sum_exp = exp_logits.sum(dim=-1, keepdim=True)

    # Softmax
    probs = exp_logits / sum_exp

    # Cross entropy loss: -log(p[target])
    # For each position, select the probability of the target class
    batch_size, seq_len, vocab_size = logits.shape
    loss = torch.zeros(batch_size, seq_len, dtype=logits.dtype, device=logits.device)

    for b in range(batch_size):
        for s in range(seq_len):
            label = labels[b, s]
            if label != ignore_index:
                loss[b, s] = -torch.log(probs[b, s, label] + 1e-8)

    return loss

import torch
import triton
import triton.language as tl

@triton.jit
def cross_entropy_fwd_kernel(
    logits_ptr,
    labels_ptr,
    loss_ptr,
    stride_logits,
    n_cols,
    ignore_index,
    BLOCK_SIZE: tl.constexpr
):
    pid = tl.program_id(axis=0)
    
    # Offset pointers to the current row
    row_logits_ptr = logits_ptr + pid * stride_logits
    row_labels_ptr = labels_ptr + pid
    row_loss_ptr = loss_ptr + pid

    # 1. Find row_max for numerical stability
    m = -float('inf')
    for i in range(0, n_cols, BLOCK_SIZE):
        offsets = i + tl.arange(0, BLOCK_SIZE)
        mask = offsets < n_cols
        x = tl.load(row_logits_ptr + offsets, mask=mask, other=-float('inf'))
        m = tl.maximum(m, tl.max(x))

    # 2. Calculate sum_exp
    sum_exp = 0.0
    for i in range(0, n_cols, BLOCK_SIZE):
        offsets = i + tl.arange(0, BLOCK_SIZE)
        mask = offsets < n_cols
        x = tl.load(row_logits_ptr + offsets, mask=mask, other=0.0)
        sum_exp += tl.sum(tl.exp(x - m))

    # 3. Compute loss
    # loss = -log(softmax(x)) = -(x - m - log(sum_exp)) = m - x + log(sum_exp)
    # We need x at the target label
    label = tl.load(row_labels_ptr)
    
    # Check if label should be ignored
    is_valid = label != ignore_index
    
    # Load target logit safely
    # Use a safe offset (0) if label is invalid to avoid bad pointer arithmetic, 
    # and rely on the mask to prevent usage.
    safe_offset = 0
    offset_to_load = tl.where(is_valid, label, safe_offset)
    
    # We use is_valid as the mask for the load to ensure we don't access garbage if label was bad,
    # although Triton's masking primarily handles out-of-bounds relative to the tensor allocation.
    # Here we rely on the logic that is_valid will be false for ignore_index.
    target_val = tl.load(row_logits_ptr + offset_to_load, mask=is_valid, other=0.0)
    
    loss_val = m - target_val + tl.log(sum_exp)
    
    # If ignore_index, loss is 0
    final_loss = tl.where(is_valid, loss_val, 0.0)
    
    tl.store(row_loss_ptr, final_loss)

@triton.jit
def cross_entropy_bwd_kernel(
    logits_ptr,
    labels_ptr,
    grad_output_ptr,
    grad_logits_ptr,
    stride_logits,
    stride_grad_logits,
    n_cols,
    ignore_index,
    BLOCK_SIZE: tl.constexpr
):
    pid = tl.program_id(axis=0)
    
    row_logits_ptr = logits_ptr + pid * stride_logits
    row_labels_ptr = labels_ptr + pid
    row_grad_output_ptr = grad_output_ptr + pid
    row_grad_logits_ptr = grad_logits_ptr + pid * stride_grad_logits

    # Recompute row_max
    m = -float('inf')
    for i in range(0, n_cols, BLOCK_SIZE):
        offsets = i + tl.arange(0, BLOCK_SIZE)
        mask = offsets < n_cols
        x = tl.load(row_logits_ptr + offsets, mask=mask, other=-float('inf'))
        m = tl.maximum(m, tl.max(x))

    # Recompute sum_exp
    sum_exp = 0.0
    for i in range(0, n_cols, BLOCK_SIZE):
        offsets = i + tl.arange(0, BLOCK_SIZE)
        mask = offsets < n_cols
        x = tl.load(row_logits_ptr + offsets, mask=mask, other=0.0)
        sum_exp += tl.sum(tl.exp(x - m))

    # Compute gradient
    # dL/dx = grad_output * (softmax(x) - 1_{x == target})
    # softmax(x) = exp(x - m) / sum_exp
    
    label = tl.load(row_labels_ptr)
    is_valid = label != ignore_index
    grad_output = tl.load(row_grad_output_ptr)

    for i in range(0, n_cols, BLOCK_SIZE):
        offsets = i + tl.arange(0, BLOCK_SIZE)
        mask = offsets < n_cols
        
        x = tl.load(row_logits_ptr + offsets, mask=mask, other=0.0)
        
        # Softmax probability
        p = tl.exp(x - m) / sum_exp
        
        # Indicator for target class
        is_target = (offsets == label)
        should_sub = is_target & is_valid
        
        # Gradient
        dlogit = p - tl.where(should_sub, 1.0, 0.0)
        
        # If label was ignored, gradient is 0
        dlogit = tl.where(is_valid, dlogit, 0.0)
        
        val = grad_output * dlogit
        
        tl.store(row_grad_logits_ptr + offsets, val, mask=mask)
