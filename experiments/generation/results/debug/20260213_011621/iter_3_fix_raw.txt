        for b in range(num_blocks):

            offs = start + tl.arange(0, BLOCK_SIZE)
            mask = offs < vocab_size
            tl.store(row_grad_ptr + offs, 0.0, mask=mask)
        return
    
    label_int = label.to(tl.int32)
    
    max_val = float('-inf')
    for b in range(num_blocks):
        start = b * BLOCK_SIZE
        offs = start + tl.arange(0, BLOCK_SIZE)
        mask = offs < vocab_size
        x = tl.load(row_logits_ptr + offs, mask=mask, other=float('-inf')).to(tl.float32)
        block_max = tl.max(x, axis=0)
        max_val = tl.maximum(max_val, block_max)
    
    sum_exp = 0.0
    for b in range(num_blocks):
        start = b * BLOCK_SIZE
        offs = start + tl.arange(0, BLOCK_SIZE)
        mask = offs < vocab_size
        x = tl.load(row_logits_ptr + offs, mask=mask, other=float('-inf')).to(tl.float32)
        exp_x = tl.exp(x - max_val)
        exp_x = tl.where(mask, exp_x, 0.0)
        sum_exp = sum_exp + tl.sum(exp_x, axis=0)
    
    inv_sum_exp = 1.0 / sum_exp
    for b in range(num_blocks):
        start = b * BLOCK_SIZE
        offs = start + tl.arange(0, BLOCK_SIZE)
        mask = offs < vocab_size
        x = tl.load(row_logits_ptr + offs, mask=mask, other=float('-inf')).to(tl.float32)
        exp_x = tl.exp(x - max_val)
        softmax = exp_x * inv_sum_exp
        is_label = offs == label_int
        grad = tl.where(is_label, softmax - 1.0, softmax)
        grad = grad * grad_out
        grad = tl.where(mask, grad, 0.0)
        tl.store(row_grad_ptr + offs, grad, mask=mask)


class CrossEntropyFunction(torch.autograd.Function):
    @staticmethod
    def forward(ctx, logits, labels, ignore_index=-100):
        if logits.dim() == 1:
            logits = logits.unsqueeze(0)
            labels = labels.unsqueeze(0)
        
        n_rows = logits.shape[0]
        vocab_size = logits.shape[1]
        
        logits = logits.contiguous()
        labels = labels.contiguous()
        
        loss = torch.empty(n_rows, dtype=torch.float32, device=logits.device)
        
        BLOCK_SIZE = 1024
        grid = (n_rows,)
        
        forward_kernel[grid](
            logits,
            labels,
            loss,
            n_rows,
            vocab_size,
            ignore_index,
            logits.stride(0),
            labels.stride(0),
            loss.stride(0),
            BLOCK_SIZE=BLOCK_SIZE,
        )
        
        ctx.save_for_backward(logits, labels)
        ctx.ignore_index = ignore_index
        
        return loss
    
    @staticmethod
    def backward(ctx, grad_output):
        logits, labels = ctx.saved_tensors
        ignore_index = ctx.ignore_index
        
        n_rows = logits.shape[0]
        vocab_size = logits.shape[1]
        
        grad_output = grad_output.contiguous()
        grad_logits = torch.zeros_like(logits)
        
        BLOCK_SIZE = 1024
        grid = (n_rows,)
        
        backward_kernel[grid](
            grad_output,
            logits,
            labels,
            grad_logits,
            n_rows,
            vocab_size,
            ignore_index,
            grad_output.stride(0),
            logits.stride(0),
            labels.stride(0),
            grad_logits.stride(0),
            BLOCK_SIZE=BLOCK_SIZE,
        )
        
        return grad_logits, None, None


def triton_cross_entropy(logits, labels, ignore_index=-100):
    return CrossEntropyFunction.apply(logits, labels, ignore_index)
