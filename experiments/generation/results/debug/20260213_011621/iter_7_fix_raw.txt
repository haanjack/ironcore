@triton.jit
def forward_kernel(
    logits_ptr,
    labels_ptr,
    loss_ptr,
    n_rows,
    vocab_size,
    ignore_index,
    stride_logits,
    stride_labels,
    stride_loss,
    BLOCK_SIZE: tl.constexpr,
):

        return
