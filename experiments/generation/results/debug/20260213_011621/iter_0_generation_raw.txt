def _reference_softmax_cross_entropy(logits, labels, ignore_index=-1):
    """PyTorch reference: Cross entropy with softmax."""
    # Stable softmax
    logits_max = logits.max(dim=-1, keepdim=True)[0]
    logits_stable = logits - logits_max

    exp_logits = torch.exp(logits_stable)
    sum_exp = exp_logits.sum(dim=-1, keepdim=True)

    # Softmax
    probs = exp_logits / sum_exp

    # Cross entropy loss: -log(p[target])
    # For each position, select the probability of the target class
    batch_size, seq_len, vocab_size = logits.shape
    loss = torch.zeros(batch_size, seq_len, dtype=logits.dtype, device=logits.device)

    for b in range(batch_size):
        for s in range(seq_len):
            label = labels[b, s]
            if label != ignore_index:
                loss[b, s] = -torch.log(probs[b, s, label] + 1e-8)

    return loss

        x = tl.load(row_logits_ptr + offs, mask=mask, other=float('-inf')).to(tl.float32)
        exp_x = tl.exp(x - max_val)
        sum_exp = sum_exp + tl.sum(exp_x, axis=0)
    
    # Compute loss: -log(softmax[label]) = log(sum_exp) + max - x[label]
    x_label = tl.load(row_logits_ptr + label).to(tl.float32)
    loss = max_val + tl.log(sum_exp) - x_label
    
    tl.store(loss_ptr + pid * stride_loss, loss)


@triton.jit
def backward_kernel(
    grad_output_ptr,
    logits_ptr,
    labels_ptr,
    grad_logits_ptr,
    n_rows,
    vocab_size,
    ignore_index,
    stride_grad_out,
    stride_logits,
    stride_labels,
    stride_grad_logits,
    BLOCK_SIZE: tl.constexpr,
):
    pid = tl.program_id(0)
    
    if pid >= n_rows:
        return
    
    grad_out = tl.load(grad_output_ptr + pid * stride_grad_out).to(tl.float32)
    label = tl.load(labels_ptr + pid * stride_labels).to(tl.int32)
    
    row_logits_ptr = logits_ptr + pid * stride_logits
    row_grad_ptr = grad_logits_ptr + pid * stride_grad_logits
    
    num_blocks = tl.cdiv(vocab_size, BLOCK_SIZE)
    
    if label == ignore_index:
        for b in range(num_blocks):
            start = b * BLOCK_SIZE
            offs = start + tl.arange(0, BLOCK_SIZE)
            mask = offs < vocab_size
            tl.store(row_grad_ptr + offs, 0.0, mask=mask)
        return
    
    # First pass: find max
    max_val = float('-inf')
    for b in range(num_blocks):
        start = b * BLOCK_SIZE
        offs = start + tl.arange(0, BLOCK_SIZE)
        mask = offs < vocab_size
        x = tl.load(row_logits_ptr + offs, mask=mask, other=float('-inf')).to(tl.float32)
        block_max = tl.max(x, axis=0)
        max_val = tl.maximum(max_val, block_max)
    
    # Second pass: compute sum of exp(x - max)
    sum_exp = 0.0
    for b in range(num_blocks):
        start = b * BLOCK_SIZE
        offs = start + tl.arange(0, BLOCK_SIZE)
        mask = offs < vocab_size
        x = tl.load(row_logits_ptr + offs, mask=mask, other=float('-inf')).to(tl.float32)
        exp_x = tl.exp(x - max_val)
        sum_exp = sum_exp + tl.sum(exp_x, axis=0)
    
    # Third pass: compute and store gradient
    for b in range(num_blocks):
        start = b * BLOCK_SIZE
        offs = start + tl.arange(0, BLOCK_SIZE)
        mask = offs < vocab_size
        x = tl.load(row_logits_ptr + offs, mask=mask, other=float('-inf')).to(tl.float32)
        softmax = tl.exp(x - max_val) / sum_exp
        is_target = tl.where(offs == label, 1.0, 0.0)
        grad = grad_out * (softmax - is_target)
        tl.store(row_grad_ptr + offs, grad, mask=mask)


class CrossEntropyFunction(torch.autograd.Function):
    @staticmethod
    def forward(ctx, logits, labels, ignore_index=-1):
        original_shape = logits.shape[:-1]
        logits_reshaped = logits.reshape(-1, logits.shape[-1]).contiguous()
        labels_reshaped = labels.reshape(-1).contiguous()
        
        n_rows, vocab_size = logits_reshaped.shape
        loss = torch.empty(n_rows, dtype=logits.dtype, device=logits.device)
        
        BLOCK_SIZE = triton.next_power_of_2(vocab_size)
        BLOCK_SIZE = max(BLOCK_SIZE, 64)
        BLOCK_SIZE = min(BLOCK_SIZE, 4096)
        
        grid = (n_rows,)
        forward_kernel[grid](
            logits_reshaped,
            labels_reshaped,
            loss,
            n_rows,
            vocab_size,
            ignore_index,
            logits_reshaped.stride(0),
            labels_reshaped.stride(0),
            loss.stride(0),
            BLOCK_SIZE=BLOCK_SIZE,
        )
        
        ctx.save_for_backward(logits_reshaped, labels_reshaped)
        ctx.ignore_index = ignore_index
        ctx.vocab_size = vocab_size
        ctx.original_shape = original_shape
        
        return loss.reshape(original_shape)
    
    @staticmethod
    def backward(ctx, grad_output):
        logits, labels = ctx.saved_tensors
        ignore_index = ctx.ignore_index
        vocab_size = ctx.vocab_size
        original_shape = ctx.original_shape
        
        grad_output_reshaped = grad_output.reshape(-1).contiguous()
        grad_logits = torch.empty_like(logits)
        
        n_rows = logits.shape[0]
        
        BLOCK_SIZE = triton.next_power_of_2(vocab_size)
        BLOCK_SIZE = max(BLOCK_SIZE, 64)
        BLOCK_SIZE = min(BLOCK_SIZE, 4096)
        
        grid = (n_rows,)
        backward_kernel[grid](
            grad_output_reshaped,
            logits,
            labels,
            grad_logits,
            n_rows,
            vocab_size,
            ignore_index,
            grad_output_reshaped.stride(0),
            logits.stride(0),
            labels.stride(0),
            grad_logits.stride(0),
            BLOCK_SIZE=BLOCK_SIZE,
        )
        
        return grad_logits.reshape(*original_shape, vocab_size), None, None


def triton_cross_entropy(logits, labels, ignore_index=-1):
    return CrossEntropyFunction.apply(logits, labels, ignore_index)
