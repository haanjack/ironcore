        offs = b + tl.arange(0, BLOCK_SIZE)
        mask = offs < vocab_size
        x = tl.load(row_logits_ptr + offs, mask=mask, other=float('-inf')).to(tl.float32)
        block_max = tl.max(x, 0)
        max_val = tl.maximum(max_val, block_max)
    
    # Compute sum of exp(x - max)
    sum_exp = 0.0
    for b in range(0, vocab_size, BLOCK_SIZE):
        offs = b + tl.arange(0, BLOCK_SIZE)
        mask = offs < vocab_size
        x = tl.load(row_logits_ptr + offs, mask=mask, other=float('-inf')).to(tl.float32)
        exp_x = tl.exp(x - max_val)
        exp_x = tl.where(mask, exp_x, 0.0)
        sum_exp = sum_exp + tl.sum(exp_x, 0)
    
    # Compute loss: log_sum_exp - x[label]
    x_label = tl.load(row_logits_ptr + label).to(tl.float32)
    log_sum_exp = max_val + tl.log(sum_exp)
    loss = log_sum_exp - x_label
    
    tl.store(loss_ptr + pid * stride_loss, loss)


@triton.jit
def backward_kernel(
    grad_output_ptr,
    logits_ptr,
    labels_ptr,
    grad_logits_ptr,
    n_rows,
    vocab_size,
    ignore_index,
    stride_grad_out,
    stride_logits,
    stride_labels,
    stride_grad_logits,
    BLOCK_SIZE: tl.constexpr,
):
    pid = tl.program_id(0)
    
    if pid >= n_rows:
        return
    
    grad_out = tl.load(grad_output_ptr + pid * stride_grad_out).to(tl.float32)
    label = tl.load(labels_ptr + pid * stride_labels).to(tl.int32)
    
    row_logits_ptr = logits_ptr + pid * stride_logits
    row_grad_ptr = grad_logits_ptr + pid * stride_grad_logits
    
    if label == ignore_index:
        for b in range(0, vocab_size, BLOCK_SIZE):
            offs = b + tl.arange(0, BLOCK_SIZE)
            mask = offs < vocab_size
            tl.store(row_grad_ptr + offs, 0.0, mask=mask)
        return
    
    # Find max for numerical stability
    max_val = float('-inf')
    for b in range(0, vocab_size, BLOCK_SIZE):
        offs = b + tl.arange(0, BLOCK_SIZE)
        mask = offs < vocab_size
        x = tl.load(row_logits_ptr + offs, mask=mask, other=float('-inf')).to(tl.float32)
        block_max = tl.max(x, 0)
        max_val = tl.maximum(max_val, block_max)
    
    # Compute sum of exp(x - max)
    sum_exp = 0.0
    for b in range(0, vocab_size, BLOCK_SIZE):
        offs = b + tl.arange(0, BLOCK_SIZE)
        mask = offs < vocab_size
        x = tl.load(row_logits_ptr + offs, mask=mask, other=float('-inf')).to(tl.float32)
        exp_x = tl.exp(x - max_val)
        exp_x = tl.where(mask, exp_x, 0.0)
        sum_exp = sum_exp + tl.sum(exp_x, 0)
    
    # Compute gradient: grad_out * (softmax - one_hot)
    # softmax = exp(x - max) / sum_exp
    for b in range(0, vocab_size, BLOCK_SIZE):
        offs = b + tl.arange(0, BLOCK_SIZE)
        mask = offs < vocab_size
        x = tl.load(row_logits_ptr + offs, mask=mask, other=float('-inf')).to(tl.float32)
        exp_x = tl.exp(x - max_val)
        exp_x = tl.where(mask, exp_x, 0.0)
        softmax = exp_x / sum_exp
        
        # Subtract 1 at the label position
        one_hot = tl.where(offs == label, 1.0, 0.0)
        grad = grad_out * (softmax - one_hot)
        
        tl.store(row_grad_ptr + offs, grad, mask=mask)


class CrossEntropyFunction(torch.autograd.Function):
    @staticmethod
    def forward(ctx, logits, labels, ignore_index=-100):
        logits = logits.contiguous()
        labels = labels.contiguous()
        
        original_shape = logits.shape
        if logits.dim() > 2:
            logits = logits.view(-1, logits.shape[-1])
            labels = labels.view(-1)
        
        n_rows, vocab_size = logits.shape
        loss = torch.empty(n_rows, dtype=torch.float32, device=logits.device)
        
        BLOCK_SIZE = triton.next_power_of_2(triton.cdiv(vocab_size, 4))
        BLOCK_SIZE = min(BLOCK_SIZE, 2048)
        BLOCK_SIZE = max(BLOCK_SIZE, 64)
        
        grid = (n_rows,)
        forward_kernel[grid](
            logits,
            labels,
            loss,
            n_rows,
            vocab_size,
            ignore_index,
            logits.stride(0),
            labels.stride(0),
            loss.stride(0),
            BLOCK_SIZE,
        )
        
        ctx.save_for_backward(logits, labels)
        ctx.ignore_index = ignore_index
        ctx.original_shape = original_shape
        
        return loss
    
    @staticmethod
    def backward(ctx, grad_output):
        logits, labels = ctx.saved_tensors
        ignore_index = ctx.ignore_index
        
        grad_output = grad_output.contiguous()
        
        if logits.dim() > 2:
            logits = logits.view(-1, logits.shape[-1])
            labels = labels.view(-1)
        
        n_rows, vocab_size = logits.shape
        grad_logits = torch.zeros_like(logits)
        
        BLOCK_SIZE = triton.next_power_of_2(triton.cdiv(vocab_size, 4))
        BLOCK_SIZE = min(BLOCK_SIZE, 2048)
        BLOCK_SIZE = max(BLOCK_SIZE, 64)
        
        grid = (n_rows,)
        backward_kernel[grid](
            grad_output,
            logits,
            labels,
            grad_logits,
            n_rows,
            vocab_size,
            ignore_index,
            grad_output.stride(0),
            logits.stride(0),
            labels.stride(0),
            grad_logits.stride(0),
            BLOCK_SIZE,
        )
        
        if len(ctx.original_shape) > 2:
            grad_logits = grad_logits.view(ctx.original_shape)
        
        return grad_logits, None, None


def triton_cross_entropy(logits, labels, ignore_index=-100):
    return CrossEntropyFunction.apply(logits, labels, ignore_index)
