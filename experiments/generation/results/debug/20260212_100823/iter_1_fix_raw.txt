    M,
    N,
    ignore_index: tl.constexpr,
    BLOCK_SIZE: tl.constexpr
):
    row_idx = tl.program_id(0)
    if row_idx >= M:
        return
        
    label = tl.load(labels_ptr + row_idx)
    grad_out = tl.load(grad_output_ptr + row_idx)
    
    is_ignored = (label == ignore_index)
    
    row_start = row_idx * stride_m
    
    for off in range(0, N, BLOCK_SIZE):
        cols = off + tl.arange(0, BLOCK_SIZE)
        mask = cols < N
        
        sm_offsets = row_start + cols * stride_n
        sm = tl.load(softmax_ptr + sm_offsets, mask=mask, other=0.0)
        
        # Gradient: (softmax - one_hot) * grad_output
        one_hot = tl.where(cols == label, 1.0, 0.0)
        dsm = sm - one_hot
        
        grad = dsm * grad_out
        
        # Zero out gradient for ignored indices
        grad = tl.where(is_ignored, 0.0, grad)
        
        grad_offsets = row_start + cols * stride_n
        tl.store(grad_logits_ptr + grad_offsets, grad, mask=mask)

class CrossEntropyFunction(torch.autograd.Function):
    @staticmethod
    def forward(ctx, logits, labels, ignore_index=-100):
        M, N = logits.shape
        stride_m = logits.stride(0)
        stride_n = logits.stride(1)
        
        # Allocate output tensors
        loss = torch.empty(M, dtype=logits.dtype, device=logits.device)
        softmax = torch.empty_like(logits)
        
        # Heuristic for BLOCK_SIZE
        BLOCK_SIZE = triton.next_power_of_2(N)
        if BLOCK_SIZE > 1024:
            BLOCK_SIZE = 1024
            
        grid = lambda META: (M,)
        
        # Launch forward kernel
        cross_entropy_forward_kernel[grid](
            logits, labels, loss, softmax,
            stride_m, stride_n, M, N,
            ignore_index,
            BLOCK_SIZE=BLOCK_SIZE
        )
        
        ctx.save_for_backward(softmax, labels)
        ctx.ignore_index = ignore_index
        ctx.M = M
        ctx.N = N
        ctx.stride_m = stride_m
        ctx.stride_n = stride_n
        ctx.BLOCK_SIZE = BLOCK_SIZE
        
        return loss, softmax

    @staticmethod
    def backward(ctx, grad_loss, grad_softmax):
        softmax, labels = ctx.saved_tensors
        ignore_index = ctx.ignore_index
        M = ctx.M
        N = ctx.N
        stride_m = ctx.stride_m
        stride_n = ctx.stride_n
        BLOCK_SIZE = ctx.BLOCK_SIZE
        
        grad_logits = torch.empty_like(softmax)
        
        grid = lambda META: (M,)
        
        cross_entropy_backward_kernel[grid](
            softmax, labels, grad_loss, grad_logits,
            stride_m, stride_n, M, N,
            ignore_index,
            BLOCK_SIZE=BLOCK_SIZE
        )
        
        # grad_softmax is unused as softmax is an intermediate, but required by autograd signature
        return grad_logits, None, None

triton_cross_entropy = CrossEntropyFunction.apply
