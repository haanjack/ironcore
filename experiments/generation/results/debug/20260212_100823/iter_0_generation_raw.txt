def _reference_softmax_cross_entropy(logits, labels, ignore_index=-1):
    """PyTorch reference: Cross entropy with softmax."""
    # Stable softmax
    logits_max = logits.max(dim=-1, keepdim=True)[0]
    logits_stable = logits - logits_max

    exp_logits = torch.exp(logits_stable)
    sum_exp = exp_logits.sum(dim=-1, keepdim=True)

    # Softmax
    probs = exp_logits / sum_exp

    # Cross entropy loss: -log(p[target])
    # For each position, select the probability of the target class
    batch_size, seq_len, vocab_size = logits.shape
    loss = torch.zeros(batch_size, seq_len, dtype=logits.dtype, device=logits.device)

    for b in range(batch_size):
        for s in range(seq_len):
            label = labels[b, s]
            if label != ignore_index:
                loss[b, s] = -torch.log(probs[b, s, label] + 1e-8)

    return loss

import torch
import triton
import triton.language as tl

@triton.jit
def cross_entropy_forward_kernel(
    logits_ptr, 
    labels_ptr, 
    loss_ptr, 
    softmax_ptr, 
    stride_m, 
    stride_n, 
    M, 
    N, 
    ignore_index: tl.constexpr,
    BLOCK_SIZE: tl.constexpr
):
    row_idx = tl.program_id(0)
    
    if row_idx >= M:
        return
    
    label = tl.load(labels_ptr + row_idx)
    row_start = row_idx * stride_m
    
    # Pass 1: Find Max
    m_max = -float('inf')
    for off in range(0, N, BLOCK_SIZE):
        cols = off + tl.arange(0, BLOCK_SIZE)
        mask = cols < N
        x = tl.load(logits_ptr + row_start + cols, mask=mask, other=-float('inf'))
        m_max = tl.maximum(m_max, tl.max(x))
        
    # Pass 2: Find Sum and Target Exp
    m_sum = 0.0
    target_exp = 0.0
    for off in range(0, N, BLOCK_SIZE):
        cols = off + tl.arange(0, BLOCK_SIZE)
        mask = cols < N
        x = tl.load(logits_ptr + row_start + cols, mask=mask, other=0.0)
        shifted = x - m_max
        exps = tl.exp(shifted)
        m_sum += tl.sum(exps)
        # Select probability of the target class
        target_exp += tl.sum(exps * (cols == label))
        
    # Pass 3: Write Softmax (Normalized)
    inv_sum = 1.0 / (m_sum + 1e-8)
    sm_row_start = row_idx * stride_m
    
    for off in range(0, N, BLOCK_SIZE):
        cols = off + tl.arange(0, BLOCK_SIZE)
        mask = cols < N
        x = tl.load(logits_ptr + row_start + cols, mask=mask, other=0.0)
        shifted = x - m_max
        exps = tl.exp(shifted)
        probs = exps * inv_sum
        tl.store(softmax_ptr + sm_row_start + cols, probs, mask=mask)
