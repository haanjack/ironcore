    log_sum_exp = max_val + tl.log(sum_exp)
    target_logit = tl.load(row_logits_ptr + label)
    loss = log_sum_exp - target_logit
    
    tl.store(loss_ptr + row_idx, loss)


@triton.jit
def backward_kernel(
    logits_ptr,
    labels_ptr,
    grad_output_ptr,
    grad_logits_ptr,
    n_rows,
    vocab_size,
    stride_logits,
    ignore_index,
    BLOCK_SIZE: tl.constexpr,
):
    row_idx = tl.program_id(0)
    
    if row_idx >= n_rows:
        return
    
    label = tl.load(labels_ptr + row_idx)
    
    if label == ignore_index:
        for block_start in range(0, vocab_size, BLOCK_SIZE):
            offsets = block_start + tl.arange(0, BLOCK_SIZE)
            mask = offsets < vocab_size
            tl.store(grad_logits_ptr + row_idx * stride_logits + offsets, 0.0, mask=mask)
        return
    
    grad_output = tl.load(grad_output_ptr + row_idx)
    row_logits_ptr = logits_ptr + row_idx * stride_logits
    
    # First pass: find max for numerical stability
    max_val = float('-inf')
    for block_start in range(0, vocab_size, BLOCK_SIZE):
        offsets = block_start + tl.arange(0, BLOCK_SIZE)
        mask = offsets < vocab_size
        logits = tl.load(row_logits_ptr + offsets, mask=mask, other=float('-inf'))
        block_max = tl.max(logits, axis=0)
        max_val = tl.maximum(max_val, block_max)
    
    # Second pass: compute sum of exponentials
    sum_exp = 0.0
    for block_start in range(0, vocab_size, BLOCK_SIZE):
        offsets = block_start + tl.arange(0, BLOCK_SIZE)
        mask = offsets < vocab_size
        logits = tl.load(row_logits_ptr + offsets, mask=mask, other=float('-inf'))
        exp_logits = tl.exp(logits - max_val)
        sum_exp += tl.sum(exp_logits, axis=0)
    
    # Third pass: compute and store gradients
    # Gradient is softmax - one_hot(label)
    # grad_logits[i] = (exp(logit_i) / sum_exp) - (1 if i == label else 0)
    # Then multiply by grad_output
    for block_start in range(0, vocab_size, BLOCK_SIZE):
        offsets = block_start + tl.arange(0, BLOCK_SIZE)
        mask = offsets < vocab_size
        logits = tl.load(row_logits_ptr + offsets, mask=mask, other=float('-inf'))
        softmax = tl.exp(logits - max_val) / sum_exp
        grad = softmax - (offsets == label).to(tl.float32)
        tl.store(grad_logits_ptr + row_idx * stride_logits + offsets, grad_output * grad, mask=mask)


class TritonCrossEntropy(torch.autograd.Function):
    @staticmethod
    def forward(ctx, logits, labels, ignore_index=-100):
        n_rows, vocab_size = logits.shape
        loss = torch.empty(n_rows, dtype=logits.dtype, device=logits.device)
        
        BLOCK_SIZE = 1024
        grid = (n_rows,)
        
        forward_kernel[grid](
            logits,
            labels,
            loss,
            n_rows,
            vocab_size,
            logits.stride(0),
            ignore_index,
            BLOCK_SIZE,
        )
        
        ctx.save_for_backward(logits, labels)
        ctx.ignore_index = ignore_index
        ctx.vocab_size = vocab_size
        
        return loss
    
    @staticmethod
    def backward(ctx, grad_output):
        logits, labels = ctx.saved_tensors
        n_rows = logits.shape[0]
        
        grad_logits = torch.zeros_like(logits)
        
        BLOCK_SIZE = 1024
        grid = (n_rows,)
        
        backward_kernel[grid](
            logits,
            labels,
            grad_output,
            grad_logits,
            n_rows,
            ctx.vocab_size,
            logits.stride(0),
            ctx.ignore_index,
            BLOCK_SIZE,
        )
        
        return grad_logits, None, None


def triton_cross_entropy(logits, labels, ignore_index=-100):
    return TritonCrossEntropy.apply(logits, labels, ignore_index)
