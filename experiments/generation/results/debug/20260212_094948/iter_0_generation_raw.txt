import torch
import triton
import triton.language as tl

@triton.jit
def forward_kernel(
    x_ptr, y_ptr, w_ptr, b_ptr, mean_ptr, rstd_ptr,
    stride, N, eps,
    BLOCK_SIZE: tl.constexpr
):
    row_idx = tl.program_id(0)
    x_ptr += row_idx * stride
    y_ptr += row_idx * stride
    
    # Compute mean
    mean = 0.0
    _mean = tl.zeros([BLOCK_SIZE], dtype=tl.float32)
    for off in range(0, N, BLOCK_SIZE):
        cols = off + tl.arange(0, BLOCK_SIZE)
        mask = cols < N
        x = tl.load(x_ptr + cols, mask=mask, other=0.).to(tl.float32)
        _mean += x
    mean = tl.sum(_mean) / N

    # Compute variance
    _var = tl.zeros([BLOCK_SIZE], dtype=tl.float32)
    for off in range(0, N, BLOCK_SIZE):
        cols = off + tl.arange(0, BLOCK_SIZE)
        mask = cols < N
        x = tl.load(x_ptr + cols, mask=mask, other=0.).to(tl.float32)
        x_centered = x - mean
        _var += x_centered * x_centered
    var = tl.sum(_var) / N
    rstd = 1.0 / tl.sqrt(var + eps)

    # Write mean and rstd for backward
    tl.store(mean_ptr + row_idx, mean)
    tl.store(rstd_ptr + row_idx, rstd)

    # Normalize and scale/shift
    for off in range(0, N, BLOCK_SIZE):
        cols = off + tl.arange(0, BLOCK_SIZE)
        mask = cols < N
        x = tl.load(x_ptr + cols, mask=mask, other=0.).to(tl.float32)
        w = tl.load(w_ptr + cols, mask=mask, other=0.).to(tl.float32)
        b = tl.load(b_ptr + cols, mask=mask, other=0.).to(tl.float32)

        tl.store(y_ptr + cols, y, mask=mask)

@triton.jit
def backward_kernel(
    x_ptr, dy_ptr, dx_ptr, dw_ptr, db_ptr, w_ptr,
    mean_ptr, rstd_ptr,
    stride, N,
    BLOCK_SIZE: tl.constexpr
):
    row_idx = tl.program_id(0)
    x_ptr += row_idx * stride
    dy_ptr += row_idx * stride
    dx_ptr += row_idx * stride

    mean = tl.load(mean_ptr + row_idx)
    rstd = tl.load(rstd_ptr + row_idx)

    # Compute sums for gradients
    sum_dy = 0.0
    sum_dy_xhat = 0.0
    _sum_dy = tl.zeros([BLOCK_SIZE], dtype=tl.float32)
    _sum_dy_xhat = tl.zeros([BLOCK_SIZE], dtype=tl.float32)

    for off in range(0, N, BLOCK_SIZE):
        cols = off + tl.arange(0, BLOCK_SIZE)
        mask = cols < N
        x = tl.load(x_ptr + cols, mask=mask, other=0.).to(tl.float32)
        dy = tl.load(dy_ptr + cols, mask=mask, other=0.).to(tl.float32)
        w = tl.load(w_ptr + cols, mask=mask, other=0.).to(tl.float32)
        
        x_hat = (x - mean) * rstd
        dy_hat = dy * w
        
        _sum_dy += dy_hat
        _sum_dy_xhat += dy_hat * x_hat

    sum_dy = tl.sum(_sum_dy)
    sum_dy_xhat = tl.sum(_sum_dy_xhat)

    # Compute dx and accumulate dw/db
    for off in range(0, N, BLOCK_SIZE):
        cols = off + tl.arange(0, BLOCK_SIZE)
        mask = cols < N
        x = tl.load(x_ptr + cols, mask=mask, other=0.).to(tl.float32)
        dy = tl.load(dy_ptr + cols, mask=mask, other=0.).to(tl.float32)
        w = tl.load(w_ptr + cols, mask=mask, other=0.).to(tl.float32)

        x_hat = (x - mean) * rstd
        dy_hat = dy * w

        dx = (dy_hat - (sum_dy + x_hat * sum_dy_xhat) / N) * rstd
        tl.store(dx_ptr + cols, dx, mask=mask)

        # Atomic adds for weight and bias gradients
        dw = dy * x_hat
        db = dy
        
        # Cast to pointer type for atomic add
        tl.atomic_add(dw_ptr + cols, dw.to(dw_ptr.dtype.element_ty), mask=mask)
        tl.atomic_add(db_ptr + cols, db.to(db_ptr.dtype.element_ty), mask=mask)
