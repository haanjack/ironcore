import triton
import triton.language as tl

@triton.jit
def rmsnorm_forward_kernel(
    x_ptr, weight_ptr, output_ptr,
    stride, n_cols, eps,
    BLOCK_SIZE: tl.constexpr,
):
    row_idx = tl.program_id(0)
    col_offsets = tl.arange(0, BLOCK_SIZE)
    mask = col_offsets < n_cols
    
    # Adjust pointers for the current row
    x_ptr += row_idx * stride
    output_ptr += row_idx * stride
    
    # Load input
    x = tl.load(x_ptr + col_offsets, mask=mask, other=0.0)
    
    # Compute variance and scale
    x2 = tl.pow(x, 2)
    var = tl.sum(x2, axis=0) / n_cols
    scale = tl.rsqrt(var + eps)
    
    # Load weight
    weight = tl.load(weight_ptr + col_offsets, mask=mask, other=0.0)
    
    # Compute output
    output = x * scale * weight
    
    # Store output
    tl.store(output_ptr + col_offsets, output, mask=mask)


@triton.jit
def rmsnorm_backward_kernel(
    x_ptr, weight_ptr, doutput_ptr,
    dx_ptr, dweight_ptr,
    stride, n_cols, eps,
    BLOCK_SIZE: tl.constexpr,
):
    row_idx = tl.program_id(0)
    col_offsets = tl.arange(0, BLOCK_SIZE)
    mask = col_offsets < n_cols
    
    # Adjust pointers for the current row
    x_ptr += row_idx * stride
    doutput_ptr += row_idx * stride
    dx_ptr += row_idx * stride
    
    # Load inputs
    x = tl.load(x_ptr + col_offsets, mask=mask, other=0.0)
    doutput = tl.load(doutput_ptr + col_offsets, mask=mask, other=0.0)
    weight = tl.load(weight_ptr + col_offsets, mask=mask, other=0.0)

    x2 = tl.pow(x, 2)
    var = tl.sum(x2, axis=0) / n_cols
    scale = tl.rsqrt(var + eps)
