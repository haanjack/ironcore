    stride_grad_x_row, stride_grad_x_col,
    n_rows, n_cols,
    eps,
    BLOCK_SIZE: tl.constexpr
):
    row_idx = tl.program_id(0)
    
    # Reduction pass for variance and ss (sum(x * w * dy))
    var = 0.0
    ss = 0.0
    
    for offset in range(0, n_cols, BLOCK_SIZE):
        cols = offset + tl.arange(0, BLOCK_SIZE)
        mask = cols < n_cols
        
        x_ptrs = x_ptr + row_idx * stride_x_row + cols * stride_x_col
        weight_ptrs = weight_ptr + cols * stride_weight_col
        grad_output_ptrs = grad_output_ptr + row_idx * stride_grad_output_row + cols * stride_grad_output_col
        
        x = tl.load(x_ptrs, mask=mask, other=0.0)
        weight = tl.load(weight_ptrs, mask=mask, other=0.0)
        grad_output = tl.load(grad_output_ptrs, mask=mask, other=0.0)
        
        var += tl.sum(x * x, axis=0)
        ss += tl.sum(x * weight * grad_output, axis=0)
        
    var /= n_cols
    inv_std = tl.rsqrt(var + eps)
    
    # dx = inv_std * (dy * w) - (x * inv_std^3 * ss) / n_cols
    factor = inv_std * inv_std * inv_std * ss / n_cols
    
    # Write pass for dx and dw
    for offset in range(0, n_cols, BLOCK_SIZE):
        cols = offset + tl.arange(0, BLOCK_SIZE)
        mask = cols < n_cols
        
        x_ptrs = x_ptr + row_idx * stride_x_row + cols * stride_x_col
        weight_ptrs = weight_ptr + cols * stride_weight_col
        grad_output_ptrs = grad_output_ptr + row_idx * stride_grad_output_row + cols * stride_grad_output_col
        grad_x_ptrs = grad_x_ptr + row_idx * stride_grad_x_row + cols * stride_grad_x_col
        grad_weight_ptrs = grad_weight_ptr + cols
        
        x = tl.load(x_ptrs, mask=mask, other=0.0)
        weight = tl.load(weight_ptrs, mask=mask, other=0.0)
        grad_output = tl.load(grad_output_ptrs, mask=mask, other=0.0)
        
        dx = inv_std * grad_output * weight - x * factor
        dw = grad_output * x * inv_std
        
        tl.store(grad_x_ptrs, dx, mask=mask)
        tl.atomic_add(grad_weight_ptrs, dw, mask=mask)

class RMSNormFunction(torch.autograd.Function):
    @staticmethod
    def forward(ctx, x, weight, eps):
        original_shape = x.shape
        x_reshaped = x.reshape(-1, x.shape[-1]).contiguous()
        
        M, N = x_reshaped.shape
        output = torch.empty_like(x_reshaped)
        
        BLOCK_SIZE = triton.next_power_of_2(N)
        if BLOCK_SIZE > 8192:
            BLOCK_SIZE = 8192
        
        grid = lambda META: (M, )
        
        rmsnorm_forward_kernel[grid](
            x_reshaped, weight, output,
            x_reshaped.stride(0), x_reshaped.stride(1),
            weight.stride(0),
            M, N, eps,
            BLOCK_SIZE=BLOCK_SIZE
        )
        
        ctx.save_for_backward(x_reshaped, weight)
        ctx.eps = eps
        ctx.original_shape = original_shape
        ctx.BLOCK_SIZE = BLOCK_SIZE
        return output.reshape(original_shape)

    @staticmethod
    def backward(ctx, grad_output):
        x_reshaped, weight = ctx.saved_tensors
        eps = ctx.eps
        original_shape = ctx.original_shape
        BLOCK_SIZE = ctx.BLOCK_SIZE
        
        grad_output_reshaped = grad_output.reshape(-1, grad_output.shape[-1]).contiguous()
        M, N = x_reshaped.shape
        
        grad_x = torch.empty_like(x_reshaped)
        grad_weight = torch.zeros_like(weight)
        
        grid = lambda META: (M, )
        
        rmsnorm_backward_kernel[grid](
            x_reshaped, weight, grad_output_reshaped,
            grad_x, grad_weight,
            x_reshaped.stride(0), x_reshaped.stride(1),
            weight.stride(0),
            grad_output_reshaped.stride(0), grad_output_reshaped.stride(1),
            grad_x.stride(0), grad_x.stride(1),
            M, N, eps,
            BLOCK_SIZE=BLOCK_SIZE
        )
        
        return grad_x.reshape(original_shape), grad_weight, None

def triton_rmsnorm(x, weight, eps=1e-5):
    return RMSNormFunction.apply(x, weight, eps)
