@triton.jit
def forward_kernel(
    x_ptr, y_ptr, w_ptr, b_ptr,
    mean_ptr, rstd_ptr,
    stride, N, eps,
    BLOCK_SIZE: tl.constexpr
):
    row = tl.program_id(0)
    row_start = row * stride
    
    # Compute Mean
    # Accumulate in float32 for precision
    _mean = tl.zeros([BLOCK_SIZE], dtype=tl.float32)
    for off in range(0, N, BLOCK_SIZE):
        col = off + tl.arange(0, BLOCK_SIZE)
        mask = col < N
        x = tl.load(x_ptr + row_start + col, mask=mask, other=0.0).to(tl.float32)
        _mean += x
    mean = tl.sum(_mean) / N
    
    # Compute Variance and Rstd
    _m2 = tl.zeros([BLOCK_SIZE], dtype=tl.float32)
    for off in range(0, N, BLOCK_SIZE):

        x = tl.load(x_ptr + row_start + col, mask=mask, other=0.0).to(tl.float32)

        _m2 += diff * diff

@triton.jit
def backward_kernel(
    x_ptr, dy_ptr, dx_ptr,
    mean_ptr, rstd_ptr, w_ptr,
    dw_ptr, db_ptr,
    stride, N,
    BLOCK_SIZE: tl.constexpr

    row_start = row * stride
    
    mean = tl.load(mean_ptr + row).to(tl.float32)
    rstd = tl.load(rstd_ptr + row).to(tl.float32)

    sum_dhat = tl.zeros([BLOCK_SIZE], dtype=tl.float32)
    sum_dhat_xhat = tl.zeros([BLOCK_SIZE], dtype=tl.float32)

    for off in range(0, N, BLOCK_SIZE):

        sum_dhat_xhat += dhat * x_hat

        tl.atomic_add(db_ptr + col, dy, mask=mask)
