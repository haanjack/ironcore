        return
    
    max_val = float('-inf')
    for block_start in range(0, D, BLOCK_D):
        offsets = block_start + tl.arange(0, BLOCK_D)
        mask = offsets < D
        logits = tl.load(logits_ptr + row_start + offsets, mask=mask, other=float('-inf'))
        block_max = tl.max(logits, 0)
        max_val = tl.maximum(max_val, block_max)
    
    sum_exp = 0.0
    for block_start in range(0, D, BLOCK_D):
        offsets = block_start + tl.arange(0, BLOCK_D)
        mask = offsets < D
        logits = tl.load(logits_ptr + row_start + offsets, mask=mask, other=float('-inf'))
        exp_vals = tl.exp(logits - max_val)
        exp_vals = tl.where(mask, exp_vals, 0.0)
        sum_exp = sum_exp + tl.sum(exp_vals, 0)
    
    for block_start in range(0, D, BLOCK_D):
        offsets = block_start + tl.arange(0, BLOCK_D)
        mask = offsets < D
        logits = tl.load(logits_ptr + row_start + offsets, mask=mask, other=float('-inf'))
        softmax = tl.exp(logits - max_val) / sum_exp
        softmax = tl.where(mask, softmax, 0.0)
        grad = tl.where(offsets == label, softmax - 1.0, softmax)
        grad = grad * grad_output
        tl.store(grad_logits_ptr + row_start + offsets, grad, mask=mask)


class CrossEntropyFunction(torch.autograd.Function):
    @staticmethod
    def forward(ctx, logits, labels, ignore_index=-100):
        N, D = logits.shape
        loss = torch.empty(N, dtype=logits.dtype, device=logits.device)
        
        BLOCK_D = min(triton.next_power_of_2(D), 4096)
        grid = (N,)
        
        forward_kernel[grid](
            logits,
            labels,
            loss,
            N,
            D,
            ignore_index,
            BLOCK_D=BLOCK_D,
        )
        
        ctx.save_for_backward(logits, labels)
        ctx.ignore_index = ignore_index
        return loss
    
    @staticmethod
    def backward(ctx, grad_output):
        logits, labels = ctx.saved_tensors
        N, D = logits.shape
        grad_logits = torch.empty_like(logits)
        
        BLOCK_D = min(triton.next_power_of_2(D), 4096)
        grid = (N,)
        
        backward_kernel[grid](
            logits,
            labels,
            grad_output,
            grad_logits,
            N,
            D,
            ctx.ignore_index,
            BLOCK_D=BLOCK_D,
        )
        
        return grad_logits, None, None


def triton_cross_entropy(logits, labels, ignore_index=-100):
    return CrossEntropyFunction.apply(logits, labels, ignore_index)
