        return

    if label == ignore_index:
        tl.store(loss_ptr + row_idx, 0.0)
        return
    
    row_start = row_idx * D
    
    max_val = float('-inf')
    for block_start in range(0, D, BLOCK_D):
        offsets = block_start + tl.arange(0, BLOCK_D)
        mask = offsets < D
        logits = tl.load(logits_ptr + row_start + offsets, mask=mask, other=float('-inf'))
        block_max = tl.max(logits, 0)
        max_val = tl.maximum(max_val, block_max)
    
    sum_exp = 0.0
    for block_start in range(0, D, BLOCK_D):
        offsets = block_start + tl.arange(0, BLOCK_D)
        mask = offsets < D
        logits = tl.load(logits_ptr + row_start + offsets, mask=mask, other=float('-inf'))
        exp_vals = tl.exp(logits - max_val)
        exp_vals = tl.where(mask, exp_vals, 0.0)
        sum_exp = sum_exp + tl.sum(exp_vals, 0)
    
    log_sum_exp = max_val + tl.log(sum_exp)
    label_logit = tl.load(logits_ptr + row_start + label)
    loss = log_sum_exp - label_logit
    tl.store(loss_ptr + row_idx, loss)


@triton.jit
def backward_kernel(
    logits_ptr,
    labels_ptr,
    grad_output_ptr,
    grad_logits_ptr,
    N,
    D,
    ignore_index,
    BLOCK_D: tl.constexpr,
):
    row_idx = tl.program_id(0)
    if row_idx >= N:
        return
    
    label = tl.load(labels_ptr + row_idx).to(tl.int32)
    row_start = row_idx * D
    grad_output = tl.load(grad_output_ptr + row_idx)
    
    if label == ignore_index:
        for block_start in range(0, D, BLOCK_D):
            offsets = block_start + tl.arange(0, BLOCK_D)
            mask = offsets < D
            tl.store(grad_logits_ptr + row_start + offsets, 0.0, mask=mask)
        return
    
    max_val = float('-inf')
    for block_start in range(0, D, BLOCK_D):
        offsets = block_start + tl.arange(0, BLOCK_D)
        mask = offsets < D
        logits = tl.load(logits_ptr + row_start + offsets, mask=mask, other=float('-inf'))
        block_max = tl.max(logits, 0)
        max_val = tl.maximum(max_val, block_max)
    
    sum_exp = 0.0
    for block_start in range(0, D, BLOCK_D):
        offsets = block_start + tl.arange(0, BLOCK_D)
        mask = offsets < D
        logits = tl.load(logits_ptr + row_start + offsets, mask=mask, other=float('-inf'))
        exp_vals = tl.exp(logits - max_val)
        exp_vals = tl.where(mask, exp_vals, 0.0)
        sum_exp = sum_exp + tl.sum(exp_vals, 0)
    
    for block_start in range(0, D, BLOCK_D):
        offsets = block_start + tl.arange(0, BLOCK_D)
        mask = offsets < D
        logits = tl.load(logits_ptr + row_start + offsets, mask=mask, other=float('-inf'))
        softmax = tl.exp(logits - max_val) / sum_exp
        softmax = tl.where(mask, softmax, 0.0)
        grad = softmax - (offsets == label).to(tl.float32)
        grad = grad * grad_output
        tl.store(grad_logits_ptr + row_start + offsets, grad, mask=mask)


class CrossEntropyFunction(torch.autograd.Function):
    @staticmethod
    def forward(ctx, logits, labels, ignore_index=-100):
        N, D = logits.shape
        loss = torch.empty(N, dtype=logits.dtype, device=logits.device)
        
        BLOCK_D = min(4096, triton.next_power_of_2(D))
        grid = (N,)
        
        forward_kernel[grid](
            logits,
            labels,
            loss,
            N,
            D,
            ignore_index,
            BLOCK_D,
        )
        
        ctx.save_for_backward(logits, labels)
        ctx.D = D
        ctx.ignore_index = ignore_index
        ctx.BLOCK_D = BLOCK_D
        return loss
    
    @staticmethod
    def backward(ctx, grad_output):
        logits, labels = ctx.saved_tensors
        N = logits.shape[0]
        grad_logits = torch.empty_like(logits)
        
        grid = (N,)
        
        backward_kernel[grid](
            logits,
            labels,
            grad_output,
            grad_logits,
            N,
            ctx.D,
            ctx.ignore_index,
            ctx.BLOCK_D,
        )
        
        return grad_logits, None, None


def triton_cross_entropy(logits, labels, ignore_index=-100):
    return CrossEntropyFunction.apply(logits, labels, ignore_index)
