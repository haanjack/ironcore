@triton.jit
def forward_kernel(
    logits_ptr,
    labels_ptr,
    loss_ptr,
    N,
    D,
    ignore_index,
    BLOCK_D: tl.constexpr,
):
    row_idx = tl.program_id(0)
    if row_idx >= N:
        return
    
    label = tl.load(labels_ptr + row_idx).to(tl.int32)
    
    if label == ignore_index:
        tl.store(loss_ptr + row_idx, 0.0)
        return
    
    row_start = row_idx * D
    
    # Find max for numerical stability
    max_val = float('-inf')
    for i in range(0, D, BLOCK_D):
        offsets = i + tl.arange(0, BLOCK_D)
        mask = offsets < D
        x = tl.load(logits_ptr + row_start + offsets, mask=mask, other=float('-inf')).to(tl.float32)
        max_val = tl.maximum(max_val, tl.max(x, axis=0))

    sum_exp = 0.0
    for i in range(0, D, BLOCK_D):
        offsets = i + tl.arange(0, BLOCK_D)
